import React from "react";
import { SurfaceState } from "../feedback/UnavailableNote";

export interface SheetProps {
  open?: boolean;
  /** Called on backdrop click, Esc, close button and drag-past-threshold. */
  onClose?: () => void;
  title?: string;
  subtitle?: string;
  /** Uppercase micro-label above the title ("SITE SURVEY"). */
  overline?: string;
  children?: React.ReactNode;
  /** auto = hugs content (max 92%); half = 56%; full = 92%. */
  size?: "auto" | "half" | "full";
  density?: "expressive" | "functional";
  /** 36x4 grab handle. Also the drag target. */
  handle?: boolean;
  /** 44x44 close button in the header. Handle is enough on mobile; use on desktop/tablet. */
  showClose?: boolean;
  /** false = must resolve via an action (destructive confirms, required steps). */
  dismissible?: boolean;
  /**
   * **Modal by default.** `false` drops the backdrop, the focus trap and the scroll lock together —
   * the three are one decision. Use it only through `EditorSurface`, whose docs carry the rule:
   * a non-modal editor is for editors whose subject is the live surface behind them (`MS3-27`'s
   * setback ring), never for a decision that must be answered.
   */
  modal?: boolean;
  dragToDismiss?: boolean;
  /** Sticky footer. Gets a luminance fade above it, never a divider line. */
  footer?: React.ReactNode;
  state?: SurfaceState;
  /** Name what failed. The default claims nothing, because a generic overlay cannot know. */
  errorTitle?: string;
  /** State the problem and the fix — and never assert a cause the surface can't verify
   *  ("The phone has no signal" on what was really a 500 is a lie the user acts on). */
  errorMessage?: string;
  onRetry?: () => void;
  /** `empty` means none **yet**, so these words invite — and `emptyAction` makes the first one. */
  emptyTitle?: string;
  emptyMessage?: string;
  emptyAction?: React.ReactNode;
  /** `unavailable` renders through `UnavailableNote`: neutral words, and **no retry, ever**. */
  unavailableTitle?: string;
  unavailableMessage?: string;
  /** One **forward** act, never a retry. */
  unavailableAction?: React.ReactNode;
  /** Position absolutely inside the nearest positioned ancestor (device frames, specimen cards). */
  inset?: boolean;
  zIndex?: number;
  labelId?: string;
  style?: React.CSSProperties;
  bodyStyle?: React.CSSProperties;
}

/**
 * Bottom sheet — the system's primary overlay. Backdrop blurs the layer behind and fades it
 * toward white at 0.35; never a dark scrim. Springs from the bottom edge, 32px top radius
 * (16px functional), e5. Traps focus, restores it on close, closes on Esc.
 *
 * **Two callers, one of them preferred.** An editor mounts `EditorSurface`, which performs
 * `F7-21`'s sheet-on-mobile / panel-on-desktop switch and renders this. Mount `Sheet` directly only
 * for a surface that is a sheet by nature at every width. With `modal={false}` the backdrop, the
 * focus trap and the scroll lock all go — see `EditorSurface` for what that variant is for.
 *
 * **Its states are the system's five** (`SurfaceState`), not a private three: a sheet is a surface,
 * so a record the market pack does not cover is `unavailable` — neutral, no retry — and a section
 * with none yet is `empty`, which invites (law 1).
 */
export function Sheet(props: SheetProps): JSX.Element | null;

export interface SheetBackdropProps {
  onClick?: () => void;
  inset?: boolean;
  zIndex?: number;
  style?: React.CSSProperties;
}

/** The backdrop alone, for surfaces that manage their own panel. */
export function SheetBackdrop(props: SheetBackdropProps): JSX.Element;

export interface SheetActionsProps {
  children?: React.ReactNode;
  /** Own-width threshold (px) below which actions stack full-width, primary on top. */
  stackBelow?: number;
  /** **The breakpoint this row owns, published** (law 4) — fires with `{stacked, width}` whenever the
      answer changes. Same callback shape as `Kanban.onFormChange` and `DataTable.onFormChange`. */
  onFormChange?: (form: { stacked: boolean; width: number }) => void;
  style?: React.CSSProperties;
}

/** Footer action row for Sheet/Modal/DetailPanel. Measures its own width, not the viewport. */
export function SheetActions(props: SheetActionsProps): JSX.Element;
