import React from "react";
import { ActionReasonSpec } from "../feedback/ActionReason";

export interface IconButtonProps {
  children?: React.ReactNode;
  /** diameter in px; visual size (min touch target stays 44) */
  size?: number;
  /** required for a11y — icon-only buttons must carry a label */
  label: string;
  variant?: "surface" | "dark" | "ghost";
  disabled?: boolean;
  /**
   * **Why it is off**, opened in the `Explainer` bubble above the circle on a tap, click, focus or
   * hover (design-system changes 5 and 6); the control becomes `aria-disabled` and stays focusable. Only with
   * `disabled`. Where the act is someone else's, the surface owes a `ScopeNote` instead.
   */
  disabledReason?: React.ReactNode | ActionReasonSpec;
  /** Holds the reason bubble open or shut; unset lets a tap open it. */
  reasonOpen?: boolean;
  style?: React.CSSProperties;
  onClick?: (e: React.MouseEvent) => void;
}

/** Perfect-circle icon button. `label` is mandatory (aria-label). */
export function IconButton(props: IconButtonProps): JSX.Element;
