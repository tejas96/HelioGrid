import React from "react";

export interface SearchFieldProps {
  value?: string;
  onChange?: (value: string) => void;
  placeholder?: string;
  density?: "expressive" | "functional";
  /** Shows a clear button once there's a value. */
  onClear?: () => void;
  disabled?: boolean;
  ariaLabel?: string;
  /** Draws the 50×50 black filter square beside the pill; opens the filter sheet. */
  onFilterClick?: () => void;
  /** Active filter count on the square — feed `FilterSet.countActive`. */
  filterCount?: number;
  /** Accessible name of the filter square. Default "Filters". */
  filterLabel?: string;
  style?: React.CSSProperties;
}

/** Grey search pill (50 tall), magnifier left, tertiary placeholder; 1.5px accent edge on focus.
 *  Pass `onFilterClick` for the black filter square beside it. */
export function SearchField(props: SearchFieldProps): JSX.Element;
