import React from "react";
import { ActionReasonSpec } from "../feedback/ActionReason";

export interface SelectOption {
  value: string;
  label: string;
  disabled?: boolean;
  /**
   * **Why it cannot be picked** — `MS2-21`'s *"disabled options explain why inline"*, `M05-53`,
   * `MS4-20`. Rendered by `ActionReason` as the **second line of the listbox row**, under the label.
   * Never a tooltip (a touch user never opens one) and never folded into the label. The row keeps its
   * `aria-disabled`, the label stays at a readable grey, and the trigger points at the walked option
   * with `aria-activedescendant` — without that last part focus never leaves the trigger and the
   * sentence would exist for everyone except the keyboard user law 9 exists for.
   */
  disabledReason?: React.ReactNode | ActionReasonSpec;
}

export interface SelectProps {
  value?: string;
  onChange?: (value: string) => void;
  /** Strings or {value,label,disabled} objects. */
  options: (SelectOption | string)[];
  label?: string;
  placeholder?: string;
  density?: "expressive" | "functional";
  disabled?: boolean;
  helper?: string;
  /** State the problem and the fix. */
  error?: string;
  name?: string;
  /**
   * The accessible name where there is **no visible `<label>`** — a `DataTable` cell editor, whose
   * visible name is the column header. Ignored when `label` is set, so the two cannot disagree.
   */
  ariaLabel?: string;
  style?: React.CSSProperties;
}

/**
 * Borderless single-select with an e4 popover listbox. Full keyboard: Up/Down/Home/End,
 * Enter or Space to open and commit, Esc to cancel, type-ahead on first letter.
 */
export function Select(props: SelectProps): JSX.Element;
