import React from "react";
import { CoachMarkAnchor } from "../feedback/CoachMark";

export interface RailItem {
  key: string;
  label: string;
  icon: React.ReactNode;
  /** Filled variant, shown only while this item is active — the one place this system mixes
   *  filled and outlined icons. Falls back to `icon`. */
  activeIcon?: React.ReactNode;
  /**
   * Unread count, read **from the record** (`F6-17`) so it always matches the list it opens.
   * Renders the numeral via `CountBadge` — "99+" above 99, nothing at zero. `true` still draws a
   * bare dot for the rare case where a count isn't known; prefer a number.
   */
  badge?: number | boolean;
  onClick?: () => void;
  /** The ref a coach mark points at to name this item. A marked item must carry one. */
  anchor?: CoachMarkAnchor;
}

export interface AppRailProps {
  items: RailItem[];
  value?: string;
  onChange?: (key: string) => void;
  /** Utility buttons pinned above the avatar (notifications, settings). */
  footer?: RailItem[];
  avatar?: React.ReactNode;
  /**
   * Rail width. Defaults to `var(--rail-w, 72px)` — the token is the source of the number, the way
   * `AppShell` reads `--header-h`. Pass a number to override it.
   */
  width?: number | string;
  /** Defaults to <LogoTile />. */
  brand?: React.ReactNode;
  style?: React.CSSProperties;
}
/** 72px desktop icon rail. Active item = accent-subtle tile, never a border or accent bar. */
export function AppRail(props: AppRailProps): JSX.Element;

export interface BottomNavProps {
  /** Five items: Home, first list, the role's verb (`verb: true`, e.g. "Add lead" / "Start survey"),
   *  second list, More. The verb takes `onClick`, never becomes active, and its label is its
   *  accessible name. `fab: true` is read as `verb: true`. */
  items: (RailItem & { verb?: boolean; fab?: boolean })[];
  value?: string;
  onChange?: (key: string) => void;
  /** Absolutely positioned over the screen, 20px from each side, 26px above the safe area. Default true. */
  floating?: boolean;
  /** Sit above the 34px home bar (--safe-bottom). Default true. */
  safeBottom?: boolean;
  style?: React.CSSProperties;
}
/** Phone footer — one floating white pill. Active item: black pill with icon + label; others:
 *  23px icons with accessible names. Replaces the arc bar and the raised FAB. */
export function BottomNav(props: BottomNavProps): JSX.Element;

export interface FabProps {
  label?: string;
  icon?: React.ReactNode;
  onClick?: () => void;
  size?: number;
  style?: React.CSSProperties;
}
/** Raised near-black action button, 56px, scales to 0.94 on press. */
export function Fab(props: FabProps): JSX.Element;
