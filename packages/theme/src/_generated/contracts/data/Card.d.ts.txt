import React from "react";
import { SurfaceState } from "../feedback/UnavailableNote";

export interface CardProps {
  children?: React.ReactNode;
  /** expressive = the tile's --r-tile and --tile-pad; functional = 12px / 16px */
  density?: "expressive" | "functional";
  /** enables the hover fill (--fill-hover) + pointer */
  interactive?: boolean;
  /** the inset 1.5px --accent ring a focused field also draws */
  selected?: boolean;
  /**
   * **The card has states**, because law 1 ("states are part of done") applies to the surface every
   * screen reaches for first. The card's frame, padding and radius never move between states — only
   * the body changes. `unavailable` renders through `UnavailableNote`: neutral, and no retry.
   */
  state?: SurfaceState;
  emptyTitle?: string;
  /** The empty sentence. Default "Nothing here yet." */
  emptyMessage?: string;
  emptyAction?: React.ReactNode;
  errorTitle?: string;
  errorMessage?: string;
  /** Draws the retry. `error` is the only state that ever offers one. */
  onRetry?: () => void;
  unavailableTitle?: string;
  unavailableMessage?: string;
  style?: React.CSSProperties;
  onClick?: (e: React.MouseEvent) => void;
}
export interface IconCircleProps {
  children?: React.ReactNode;
  /** semantic or brand colour; container fills a 6% tint of it */
  color?: string;
  size?: number;
  style?: React.CSSProperties;
}

/**
 * A grey tile (--fill) on the white page — no border, no shadow. Ships `loading`, `empty`,
 * `error` and `unavailable` states, like every other surface in the system.
 * @startingPoint section="Data display" subtitle="Floating card + signature circular icon container" viewport="700x260"
 */
export function Card(props: CardProps): JSX.Element;
/** Signature circular icon container filled with a 6% tint of its colour. */
export function IconCircle(props: IconCircleProps): JSX.Element;
