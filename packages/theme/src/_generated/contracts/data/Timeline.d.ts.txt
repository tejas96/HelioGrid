import React from "react";
import { SurfaceState } from "../feedback/UnavailableNote";

export interface TimelineItem {
  id?: string;
  label: string;
  /** Right-aligned micro text — a date, a count, a duration. */
  meta?: string;
  description?: string;
  /**
   * Who did it — renders with a small person glyph. **A free string, and that is the limit of what
   * this component claims:** it cannot distinguish a person from the agent, the system or the
   * customer, which `M07-03` requires of an activity log. An entry whose actor is not a person
   * belongs in `ActivityStream`, where the actor **class** is structural.
   */
  actor?: string;
  /** Arbitrary node under the step: photos, a StatCard pair, an action. */
  content?: React.ReactNode;
  status?: "done" | "current" | "upcoming" | "blocked" | "failed";
}

export interface TimelineProps {
  items: TimelineItem[];
  /** page = full-page sequence with a continuous rail; compact = the four-row sheet form. */
  variant?: "page" | "compact";
  density?: "expressive" | "functional";
  state?: SurfaceState;
  emptyTitle?: string;
  emptyDescription?: string;
  errorTitle?: string;
  errorMessage?: string;
  onRetry?: () => void;
  /** `unavailable`: there is no sequence here to draw. Neutral, and no retry. */
  unavailableTitle?: string;
  unavailableMessage?: string;
  style?: React.CSSProperties;
}

/**
 * Vertical sequence of steps. A continuous rail runs through the nodes — lit to the current
 * step, dim beyond it — so it reads as a sequence, not a list of dots. Done = filled success
 * tick, current = accent dot in a soft ring, upcoming = hollow and tertiary.
 *
 * **This is a sequence, not a stream.** An append-only activity log — 13+ kinds, four actor classes,
 * hundreds of entries, every one of them `done` — is `ActivityStream`: the rail would light
 * everything and say nothing, and there would be no current step to be dim beyond. `SCR-M08-02`
 * needs both on one screen, one directly beneath the other.
 *
 * **And it reports; it does not collect.** No item here has a toggle: work still to be done, ticked off
 * in any order with materials and attribution, is `Checklist` (law 23). A severity-ranked set of gate
 * failures is `FindingList` (law 20) — it has no current step for the rail to light.
 */
export function Timeline(props: TimelineProps): JSX.Element;
