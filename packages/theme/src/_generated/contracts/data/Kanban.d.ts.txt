import React from "react";
import { SolarStatus } from "./StatusChip";
import { PendingActionSpec } from "../feedback/PendingAction";
import { SurfaceState } from "../feedback/UnavailableNote";
import { ProvenanceProps, ProvenanceTierSpec, ProvenanceStanding } from "./Provenance";

export interface KanbanCardItem {
  id: string | number;
  title: string;
  /** Second line — city, capacity, due date. */
  meta?: string;
  /** Right-aligned mono value, e.g. ₹5,12,000. */
  value?: string;
  /**
   * **The tier on this card's figure** (`F8-01` / `F8-07`) — a bare tier word, a full spec or a ready
   * node, rendered as persistent words on their own line under the meta/value row.
   *
   * A board card is a record carrying money, and the board is not an exception to the tier rule: this
   * is the same slot `DataTable`'s row and stacked record card fill, so one record reads alike in all
   * three forms. Before it existed the only route was `renderCard` — a private adjacency per screen,
   * which is what `StatCard`'s slot rule forbids.
   */
  provenance?: ProvenanceProps | ProvenanceTierSpec;
  /** How far the figure can be relied on as final. Folded into `provenance` when both are given. */
  standing?: ProvenanceStanding;
  /** Name for the trailing avatar. */
  owner?: string;
  [key: string]: any;
}

export interface KanbanColumn {
  key: string;
  label: string;
  /** Drives the header StatusChip. */
  status?: SolarStatus;
  items?: KanbanCardItem[];
  /** WIP limit — header shows n/limit and turns warning when exceeded. */
  limit?: number;
  emptyLabel?: string;
}

export interface KanbanProps {
  columns: KanbanColumn[];
  onCardClick?: (item: KanbanCardItem, column: KanbanColumn) => void;
  /** Enables drag between columns. Called with (id, toColumnKey, fromColumnKey). */
  onMove?: (id: string | number, to: string, from: string) => void;
  /**
   * Replaces the whole card. **Not the route for a tier or a standing** — those are `item.provenance`
   * and `item.standing`, so the adjacency stays the system's rather than each screen's.
   */
  renderCard?: (item: KanbanCardItem, column: KanbanColumn) => React.ReactNode;
  /**
   * **This card's move is in flight** (`SCR-M08-01`'s `move-waiting-server`: *"the move waits visibly
   * until it reaches the server"*). Return a `PendingActionSpec` or null; `PendingAction` renders it
   * under the card. The card is **not** dimmed, disabled or covered — it stays draggable, its move
   * buttons stay live, and a second move can start while the first is in flight. A failure comes back
   * as `state: "returned"` with a reason, and the card is already back where it was, because nothing
   * was optimistically applied (`F8-36`).
   */
  cardPending?: (item: KanbanCardItem, column: KanbanColumn) => PendingActionSpec | null;
  state?: SurfaceState;
  emptyTitle?: string;
  emptyDescription?: string;
  /** `unavailable`: no such board here at all. Neutral, and no retry — see `UnavailableNote`. */
  unavailableTitle?: string;
  unavailableMessage?: string;
  errorTitle?: string;
  errorMessage?: string;
  onRetry?: () => void;
  columnWidth?: number;
  /**
   * Own-width px below which the board becomes **one column with a stage filter** — `M08-10` (P0) /
   * `SCR-M08-01`'s `mobile-single-column-filter`. Default 720.
   *
   * It used to stack: nine full-width columns one after another, which at the ruled 200-project
   * volume is the whole portfolio in one scroll. The phone form is now a `FilterChips` stage strip
   * (one-of-N by construction, with a count per stage) above the selected column.
   */
  stackBelow?: number;
  /**
   * The stage shown on the phone form. Uncontrolled by default (the first column). Pass it — with
   * `onStackedColumnChange` — when the screen keeps the stage in a URL or shares it with a desktop
   * scroll position.
   */
  stackedColumn?: string;
  onStackedColumnChange?: (key: string) => void;
  /** The overline above the stage strip. Default "Stage"; pass "" to drop it. */
  stageFilterLabel?: string;
  /**
   * **The measurement, published.** Fires whenever the answer changes: `{stacked, width}` — the
   * board's own width, never the viewport. A screen arranging content around the board (a stage
   * header, a New job button, a count line) needs to know *when* the form switched, and this
   * component owns the private `ResizeObserver` that knows — so it says, rather than leaving every
   * caller to run a second observer over the same element and drift.
   *
   * `DataTable` and `SheetActions` — the law's other two named breakpoint owners — publish through
   * this exact callback. `EditorSurface` publishes the same idea in a different vocabulary, and the
   * difference is worth knowing rather than glossing: it hands back `{ref, layerWidth, panel, probeStyle}`
   * from a hook whose probe the caller mounts, because there the caller's element is what gets measured.
   */
  onFormChange?: (form: { stacked: boolean; width: number }) => void;
  style?: React.CSSProperties;
}

/**
 * Pipeline board — sunken grey columns, floating functional cards, StatusChip headers.
 * Functional density only. Each column carries its own empty state.
 *
 * **Two forms, and the component owns the switch on its own width** (law 4). Above `stackBelow` the
 * full board scrolls sideways between columns; below it, `M08-10`'s phone answer — **one column with
 * a stage filter**, not nine columns stacked. The measurement is published through `onFormChange`,
 * because a screen cannot arrange around a breakpoint it is not told about.
 *
 * A move is always available without a mouse (always-visible 44px move buttons), and on the phone
 * form they are the only route: the destination column is not on screen.
 */
export function Kanban(props: KanbanProps): JSX.Element;
